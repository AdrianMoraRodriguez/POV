public void Make(){
        

        var mesh=_retrievedMesh;


        matData=new RetrievedMaterial[mesh.materials.Length-1];
        for(int i=1;i<mesh.materials.Length;i++)
            matData[i-1]=mesh.materials[i]; // The first one is a default

        if(mesh.vertexdata is null || mesh.weightdata is null || mesh.indexdata is null)
            throw new Exception("Error: mesh data is wrong or empty");
 
        slotData=new int[mesh.materials.Length-1];
        
        int nindex=0;
        for(int i=0;i<(mesh.materials.Length-1);i++)
        {
            slotData[i]=nindex;
            nindex+=mesh.indexdata[i].Length;
        
        }

        indexData=new int[nindex];
        int count=0;
        for(int i=0;i<(mesh.materials.Length-1);i++){
            for(int j=0;j<mesh.indexdata[i].Length;j++)
                indexData[count++]=mesh.indexdata[i][j];
        }

        int nvertices=mesh.vertexdata.Length;
        int nweight=mesh.weightdata.Length;
        int nnormal=mesh.normaldata.Length;
        int nuv=mesh.uv.Length;

        if((nvertices/3) != nweight)
            throw new Exception("Number of vertex weights is different of number of vertices");

        
        vertexData=new float[nvertices+nweight+nnormal+nuv];
        int nvalues = 9; // 3 components per vers, 1 per weight
        for(int i=0,j=0,k=0,l=0,m=0;i<mesh.vertexdata.Length;i=i+3,j=j+1,l=l+2,m=m+3,k=k+nvalues){
            vertexData[k]=mesh.vertexdata[i];
            vertexData[k+1]=mesh.vertexdata[i+1];
            vertexData[k+2]=mesh.vertexdata[i+2];
            vertexData[k+3]=mesh.weightdata[j];
            vertexData[k+4]=mesh.uv[l];
            vertexData[k+5]=mesh.uv[l+1];
            vertexData[k+6]=mesh.normaldata[m];
            vertexData[k+7]=mesh.normaldata[m+1];
            vertexData[k+8]=mesh.normaldata[m+2];
        }
        

    }

// Crear la clase Collision

class Collision{

static public bool CollisionLess( Vector3 ?v1, Vector3 ?v2){
        bool identical=true;
        bool result=true;
        if(v1 is null || v2 is null)
            return true;

        for(int i=0;i<3;i++){
            if(v1.Value[i]>v2.Value[i]){
                result=false;
                break;
            }
            if(identical && v1.Value[i]<v2.Value[i])
                identical=false;
        }


        return (!identical && result);
    }

static public void CollisionMinMax(List<Vector3> list, out Vector3 ?Min, out Vector3 ?Max){
        Min=null;
        Max=null;
        foreach(Vector3 v in list){
            if(CollisionLess(v,Min)){
                Min=v;
            }
            if(CollisionLess(Max,v)){
                Max=v;
            }
        }


    }

public static bool CheckCollisionAABB(Actor actor1,Actor actor2){
            // TODO: considerar no solo la transformacion de traslacion
            bool result=false;

            Vector3 v1=actor1.Model.ExtractTranslation();
            Vector3 v2=actor2.Model.ExtractTranslation();
            List<Vector3>? actorBox1=actor1.GetCollisionBox();
            List<Vector3>? actorBox2=actor2.GetCollisionBox();
            if(actorBox1 is not null && actorBox2 is not null){

                List<Vector3> box1=new List<Vector3>();
                List<Vector3> box2=new List<Vector3>();

                // Translate each box
                // TODO: scale !!!!!d

                foreach(Vector3 v in actorBox1)
                    box1.Add(v+v1);
                
                foreach(Vector3 v in actorBox2)
                    box2.Add(v+v2);

                // Check AABB Collision
                
                if(CollisionLess(box1[0],box2[1]) && CollisionLess(box2[0],box1[1]))
                    result=true;

            }
                
            return result;
        }

} // Class

using OpenTK.Mathematics;
namespace POV{

class Utils {
    static public List<Vector3> GenerateVector3List(float [] data){

        List<Vector3> list = new List<Vector3>();

        for(int i=0;i<data.Length;i+=3){
            if(i+2>data.Length)
                break;
            list.Add(new Vector3(data[i],data[i+1],data[i+2]));
        }
        return list;

    }
}
}

// En Actor

public CollisionType CollisionType;

public List<Vector3> CollisionData = new List<Vector3>();

// En Actor

public void GenerateCollisionData(CollisionType collisionType,Mesh mesh ){
        

            CollisionData=new List<Vector3>();
            List<Vector3> vertices= Utils.GenerateVector3List(mesh.vertexData);
            Vector3 ?Min=null;
            Vector3 ?Max=null;

            Collision.CollisionMinMax(vertices,out Min, out Max);
            if(Min is not null && Max is not null){
                CollisionData.Add(Min.Value);
                CollisionData.Add(Max.Value);
            }

        

    }

    public List<Vector3>? GetCollisionBox(){
        if(CollisionData.Count<2)
            return null;
        List<Vector3> result=CollisionData.GetRange(0,2); // Design decision, return only the required vertices for a Box.
        return result;
    }

protected bool CheckPawnCollision(){
    bool result=false;
    Actor? actorPawn=null;
    if(_level.ActorCollection.ContainsKey("apawn")){
        actorPawn=_level.ActorCollection["apawn"];
      
        foreach(var actor in _level.ActorCollection){
            if(actor.Key=="apawn")
                continue;
            if(Collision.CheckCollisionAABB(actorPawn,actor.Value)){
                result=true;
                break;
            }
        }
    }
    return result;
}
